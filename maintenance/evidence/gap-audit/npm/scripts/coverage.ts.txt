import {readFile,readdir,writeFile,copyFile,mkdir} from 'node:fs/promises'
import {resolve,dirname,join} from 'node:path'
import assert from 'node:assert/strict'
import semver from 'semver'
import {digest} from '../../maintenance/model.ts'
const root=process.cwd(),out=resolve('maintenance/evidence/gap-audit/npm')
const reportPath='maintenance/evidence/compatibility-policy-update/report.json',dataPath='data/compatibility.json'
const reportRaw=await readFile(reportPath),dataRaw=await readFile(dataPath)
const report=JSON.parse(reportRaw.toString()),data=JSON.parse(dataRaw.toString())
const review=JSON.parse(await readFile(join(out,'review-audit.json'),'utf8'))
const byObs=new Map(review.observations.map((o:any)=>[o.id,o]))
const files=(await readdir('maintenance/evidence',{recursive:true})).filter((p:string)=>/\/rows\/[^/]+\.json$/.test(p))
const rowPaths=new Map(files.map((p:string)=>[p.split('/').at(-1)!.slice(0,-5),'maintenance/evidence/'+p]))
const totals:any={},fixtures=[];let logChecks=0
for(const f of report.fixtures.filter((f:any)=>f.fixtureId.startsWith('npm'))){const rule=data.rules.find((r:any)=>r.manager==='npm'&&JSON.stringify(r.match)===JSON.stringify(f.match));const points=[];for(const p of f.points){let category='supported',verifiedIds=[],rowAudit=[];if(p.outcome!=='supported'){if(p.version==='1.1.25')category='invalid-experiment-old-runtime-bootstrap';else if(semver.lt(p.version,'5.7.0'))category='required-ci-command-not-implemented';else if(semver.lt(p.version,'7.0.0-0'))category='actual-format-incompatibility-legacy-verifier-requires-dependencies-map';else if(semver.gte(p.version,'12.0.0-0'))category='actual-filename-support-removal-npm12';else category=p.outcome==='unsupported'?'upstream-ci-save-bug-1811':'upstream-ci-validation-bug-4363';
for(const id of p.rowIds){const rowPath=rowPaths.get(id)!;const raw=await readFile(rowPath);const row=JSON.parse(raw.toString());if(row.protocol!=='frozen-initial-matrix-v2')continue;const o=row.observation;if(category==='invalid-experiment-old-runtime-bootstrap'){assert.equal(row.status,'inconclusive');assert(!o);assert(/No official checksum|psl\.cjs/.test(row.error));rowAudit.push({rowId:id,rowPath,rowSha256:digest(raw),error:row.error.includes('psl.cjs')?'Modern bootstrap transitive psl syntax cannot parse on Node 0.8':'Unavailable Darwin arm64 archive for historical Node'});continue;}if(!o||!p.evidenceIds.includes(o.id))continue;if(category.startsWith('upstream-')){const reviewed:any=byObs.get(o.id);if(reviewed){verifiedIds.push(o.id);rowAudit.push({rowId:id,rowPath,rowSha256:digest(raw),observationId:o.id,reviewId:reviewed.bugId})}continue;}if(!['incompatible','semantic-mismatch'].includes(o.status))continue;const logPath=resolve(dirname(dirname(rowPath)),o.logPath);const text=await readFile(logPath,'utf8');if(category==='required-ci-command-not-implemented'){const list=text.replaceAll('\r\n','\n').split('where <command> is one of:').at(1)?.trimStart();const end=list?.indexOf('\n\n')??-1;const commands=end===-1?undefined:list?.slice(0,end);assert(/Unknown command:\s*["']?ci["']?(?:\s|$)/i.test(text)||(commands&&!commands.split(/[\s,]+/).includes('ci')),'not explicit no-ci '+o.id)}else if(category.startsWith('actual-format'))assert(/lock-verify[/\\]index\.js:\d+:\d+/.test(text)&&/Cannot read propert(?:y|ies)/.test(text),'not proven legacy verifier '+o.id);else {const normalized=text.replace(/^npm (?:ERR!|error)\s*/gm,'').replace(/\s+/g,' ');assert(/can only install with an existing package-lock\.json with lockfileVersion >= 1/.test(normalized));}logChecks++;verifiedIds.push(o.id);rowAudit.push({rowId:id,rowPath,rowSha256:digest(raw),observationId:o.id,logPath:logPath.replace(root+'/',''),logSha256:digest(text)});}
assert(category==='invalid-experiment-old-runtime-bootstrap'?rowAudit.length>0:verifiedIds.length>0,'unclassified '+f.fixtureId+' '+p.version)}
totals[category]=(totals[category]??0)+1;points.push({version:p.version,prerelease:p.prerelease,reportedOutcome:p.outcome,classification:category,admittedByShippedRange:!!rule&&semver.satisfies(p.version,rule.range),...(p.outcome!=='supported'?{verifiedEvidenceIds:verifiedIds,rows:rowAudit}:{})})}
const runs=[];for(const p of points){const prev=runs.at(-1);if(prev&&prev.classification===p.classification){prev.end=p.version;prev.count++;}else runs.push({start:p.version,end:p.version,count:1,classification:p.classification})}
fixtures.push({fixtureId:f.fixtureId,fixtureHash:f.fixtureHash,match:f.match,shippedRange:rule?.range??null,reportedRange:f.range,totalPoints:points.length,runs,points})}
const coverage={createdAt:new Date().toISOString(),inputs:{reportPath,reportSha256:digest(reportRaw),dataPath,dataSha256:digest(dataRaw)},totalFixtures:fixtures.length,totalPoints:fixtures.reduce((n,f)=>n+f.totalPoints,0),classificationCounts:totals,rawConclusiveFailureLogsChecked:logChecks,exactKnownBugReviewObservations:review.reviewCount,fixtures}
await writeFile(join(out,'coverage.json'),JSON.stringify(coverage,null,2)+'\n')
console.log(JSON.stringify({totalPoints:coverage.totalPoints,classificationCounts:totals,rawFailureLogs:logChecks},null,2))
