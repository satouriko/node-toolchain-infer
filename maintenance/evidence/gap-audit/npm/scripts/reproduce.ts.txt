import {readFile,writeFile,mkdir,copyFile,cp} from 'node:fs/promises'
import {join,resolve,dirname} from 'node:path'
import { provision } from '../../maintenance/provision.ts'
import { runFixture } from '../../maintenance/runner.ts'
import { digest } from '../../maintenance/model.ts'
const root=process.cwd()
const catalog=JSON.parse(await readFile('maintenance/evidence/yarn-pnp-matrix/catalog.json','utf8'))
const fixtures=JSON.parse(await readFile('fixtures/recipes.json','utf8')).filter((x:any)=>x.manager==='npm')
const versions=process.argv.slice(2).length?process.argv.slice(2):['7.0.0-beta.0','7.0.0-beta.5','7.0.0-beta.6','7.0.0-beta.11','7.0.0-beta.12','7.0.0','8.4.0','8.4.1']
let cursor=0
async function worker(){while(cursor<versions.length){const version=versions[cursor++];const release=catalog.managers.npm.find((x:any)=>x.version===version);const dir=resolve('.cache/npm-gap-audit/tools',version);const tool=await provision('npm',release,catalog,dir,undefined,{cacheDirectory:resolve('.cache/maintenance'),historicalNode:true});console.log('Provisioned',version,tool.nodeVersion);for(const fixture of fixtures){const out=resolve('maintenance/evidence/gap-audit/npm/reproductions',version,fixture.id);await mkdir(out,{recursive:true});const work=resolve('.cache/npm-gap-audit/work',version,fixture.id);const observation=await runFixture(fixture,tool,resolve('fixtures'),work,out);await writeFile(join(out,'observation.json'),JSON.stringify(observation,null,2)+'\n');await writeFile(join(out,'tool.json'),JSON.stringify(tool,null,2)+'\n');for(const f of [fixture.lock,'package.json']){await mkdir(join(out,'before'),{recursive:true});await mkdir(join(out,'after'),{recursive:true});await copyFile(join(root,'fixtures',fixture.directory,f),join(out,'before',f));await copyFile(join(work,f),join(out,'after',f));}const before=JSON.parse(await readFile(join(out,'before',fixture.lock),'utf8'));const after=JSON.parse(await readFile(join(out,'after',fixture.lock),'utf8'));function graph(x:any){return x.packages?Object.fromEntries(Object.entries(x.packages).filter(([p])=>p!=='').map(([p,v]:any)=>[p,{version:v.version,resolved:v.resolved,integrity:v.integrity}])):Object.fromEntries(Object.entries(x.dependencies??{}).map(([p,v]:any)=>['node_modules/'+p,{version:v.version,resolved:v.resolved,integrity:v.integrity}]))};const b=graph(before),a=graph(after);const compare={beforeFormat:before.lockfileVersion,afterFormat:after.lockfileVersion,dependencyGraphEqual:JSON.stringify(b)===JSON.stringify(a),beforeGraph:b,afterGraph:a,packageJsonParsedEqual:JSON.stringify(JSON.parse(await readFile(join(out,'before/package.json'),'utf8')))===JSON.stringify(JSON.parse(await readFile(join(out,'after/package.json'),'utf8')))};await writeFile(join(out,'semantic-comparison.json'),JSON.stringify(compare,null,2)+'\n');console.log(version,fixture.id,observation.status,'control',observation.frozenControl?.status,'semantic',observation.semantic,'graphEqual',compare.dependencyGraphEqual,'formats',compare.beforeFormat,compare.afterFormat)}}}
await Promise.all([worker(),worker()])
