import {readFile,writeFile,mkdir,copyFile} from 'node:fs/promises'
import {join,resolve} from 'node:path'
import {provision} from '../../maintenance/provision.ts'
import {runFixture,execute} from '../../maintenance/runner.ts'
const catalog=JSON.parse(await readFile('maintenance/evidence/yarn-pnp-matrix/catalog.json','utf8'))
const outRoot=resolve('maintenance/evidence/gap-audit/npm/lock-selection-canary')
const generator=JSON.parse(await readFile('maintenance/evidence/gap-audit/npm/reproductions/8.4.1/npm-package-lock-v1/tool.json','utf8'))
await mkdir(outRoot,{recursive:true})
const fixtures=[]
for(const format of [1,2,3]){const dir=join(outRoot,'fixtures',`package-lock-v${format}`);await mkdir(dir,{recursive:true});const manifest={name:'lock-selection-canary',version:'1.0.0',private:true,dependencies:{'is-number':'6.0.0'}};await writeFile(join(dir,'package.json'),JSON.stringify(manifest,null,2)+'\n');const args=['install','--package-lock-only',`--lockfile-version=${format}`,'--ignore-scripts','--no-audit','--no-fund'];const first=await execute(generator,args,dir);if(first.exitCode!==0)throw Error(first.output);manifest.dependencies['is-number']='^6.0.0 || ^7.0.0';await writeFile(join(dir,'package.json'),JSON.stringify(manifest,null,2)+'\n');const second=await execute(generator,args,dir);if(second.exitCode!==0)throw Error(second.output);const lock=JSON.parse(await readFile(join(dir,'package-lock.json'),'utf8'));const locked=lock.packages?.['node_modules/is-number']?.version??lock.dependencies?.['is-number']?.version;if(locked!=='6.0.0')throw Error('Canary generator did not retain version6');await writeFile(join(outRoot,`generation-v${format}.log`),first.output+'\n'+second.output);for(const file of ['package-lock.json','npm-shrinkwrap.json']){const id=`canary-${file.replace('.json','')}-v${format}`;const fixtureDir=join(outRoot,'clean-fixtures',id);await mkdir(fixtureDir,{recursive:true});await copyFile(join(dir,'package.json'),join(fixtureDir,'package.json'));await copyFile(join(dir,'package-lock.json'),join(fixtureDir,file));fixtures.push({id,manager:'npm',version:'8.4.1',directory:id,lock:file,match:{format,file},dependencies:{'is-number':'6.0.0'}})}}
await writeFile(join(outRoot,'fixtures.json'),JSON.stringify(fixtures,null,2)+'\n')
const versions=['7.0.0-beta.0','7.0.0-beta.6','7.0.0-beta.11','7.0.0-beta.12','7.0.0','7.24.2','8.0.0','8.4.0','8.4.1'];let cursor=0
async function worker(){while(cursor<versions.length){const version=versions[cursor++];let tool;try{tool=JSON.parse(await readFile(`maintenance/evidence/gap-audit/npm/reproductions/${version}/npm-package-lock-v1/tool.json`,'utf8'))}catch{tool=await provision('npm',catalog.managers.npm.find((r:any)=>r.version===version),catalog,resolve('.cache/npm-gap-audit/tools',version),undefined,{cacheDirectory:resolve('.cache/maintenance'),historicalNode:true})};const noLock=resolve('.cache/npm-gap-audit/canary-work',version,'without-lock');await mkdir(noLock,{recursive:true});await copyFile(join(outRoot,'clean-fixtures',fixtures[0].id,'package.json'),join(noLock,'package.json'));const baseline=await execute(tool,['install','--ignore-scripts','--no-audit','--no-fund'],noLock);const baselineInstalled=JSON.parse(await readFile(join(noLock,'node_modules/is-number/package.json'),'utf8')).version;await mkdir(join(outRoot,version),{recursive:true});await writeFile(join(outRoot,version,'without-lock.json'),JSON.stringify({command:['install','--ignore-scripts','--no-audit','--no-fund'],exitCode:baseline.exitCode,installed:baselineInstalled,node:tool.nodeVersion,output:baseline.output},null,2));for(const fixture of fixtures){const out=join(outRoot,version,fixture.id);await mkdir(out,{recursive:true});const observation=await runFixture(fixture,tool,join(outRoot,'clean-fixtures'),resolve('.cache/npm-gap-audit/canary-work',version,fixture.id),out);await writeFile(join(out,'observation.json'),JSON.stringify(observation,null,2)+'\n');console.log(version,fixture.id,observation.status,'locked',observation.installed['is-number'],'withoutLock',baselineInstalled)}}}
await Promise.all([worker(),worker()])
